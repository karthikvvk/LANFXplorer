Get system capacity:
import os

pages = os.sysconf("SC_PHYS_PAGES")
page_size = os.sysconf("SC_PAGE_SIZE")
ram_bytes = pages * page_size
value = int(ram_bytes * 0.01)
value = min(value, 2147483647)
print(value)
print(f"{value / 1024 / 1024:.2f} MB")

INSTALL

KEY GEN

            -out server.crt -days 365 -nodes -subj "/CN=localhost"

Build
cmake -B build && cmake --build build


COMPILE
g++ -std=c++17 -O2 receiver.cpp -lmsquic -o build/receiver
g++ -std=c++17 -O2 sender.cpp -lmsquic -o build/sender





RUN
# Terminal 1
./build/receiver 4433 received.bin

# Terminal 2
./build/sender testfile.img 127.0.0.1 4433



VERIFY
